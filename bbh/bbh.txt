diamond.exe makedb --in 4414.fasta -d 4414_db
diamond.exe blastp -q 4414.fasta -d iCre1355_db -o 4414_vs_iCre1355.tsv --outfmt 6 --evalue 1e-10 --max-target-seqs 1
diamond.exe blastp -q iCre1355.fasta -d 4414_db -o iCre1355_vs_4414.tsv --outfmt 6 --evalue 1e-10 --max-target-seqs 1

import pandas as pd

print("正在读取 DIAMOND 比对结果...")

# DIAMOND outfmt 6 的标准列名
columns = ['query_id', 'subject_id', 'identity', 'alignment_length', 
           'mismatches', 'gap_opens', 'q_start', 'q_end', 
           's_start', 's_end', 'evalue', 'bit_score']

# 1. 读取正向和反向比对结果
# 请确保文件名与你实际生成的文件名一致
forward_df = pd.read_csv('4414_vs_iCre1355.tsv', sep='\t', names=columns)
reverse_df = pd.read_csv('iCre1355_vs_4414.tsv', sep='\t', names=columns)

# 2. 确保每个 query 只保留得分最高的第一条结果（以防 --max-target-seqs 1 输出了多个局部比对）
forward_best = forward_df.drop_duplicates(subset=['query_id'])
reverse_best = reverse_df.drop_duplicates(subset=['query_id'])

# 3. 提取基因映射字典以加速查找
# forward_dict: {4414_gene: iCre1355_gene}
# reverse_dict: {iCre1355_gene: 4414_gene}
forward_dict = dict(zip(forward_best['query_id'], forward_best['subject_id']))
reverse_dict = dict(zip(reverse_best['query_id'], reverse_best['subject_id']))

print(f"正向比对找到 {len(forward_dict)} 个最佳匹配。")
print(f"反向比对找到 {len(reverse_dict)} 个最佳匹配。")

# 4. 寻找双向最佳比对 (BBH)
bbh_indices = []

for q_id, s_id in forward_dict.items():
    # 如果 iCre 的基因 s_id 存在于反向比对结果中，并且它匹配回的基因正是 q_id
    if s_id in reverse_dict and reverse_dict[s_id] == q_id:
        # 记录在 forward_best 中的索引
        bbh_indices.append(forward_best[forward_best['query_id'] == q_id].index[0])

# 5. 提取 BBH 对应的完整数据行 (保留 evalue, bit_score 等供后续 MATLAB 使用)
bbh_df = forward_best.loc[bbh_indices]

print(f"提取完成！共找到 {len(bbh_df)} 对双向最佳比对 (BBH)。")

# 6. 保存为 CSV 文件
output_filename = 'BBH_results.csv'
bbh_df.to_csv(output_filename, index=False)

print(f"结果已成功保存为：{output_filename}")
